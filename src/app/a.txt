import axios from "axios";

const BASE_URL =
  process.env.NEXT_PUBLIC_BASEAPI || "http://192.168.2.181:3000/admin";

// Axios instance
const apiClient = axios.create({
  baseURL: BASE_URL,
  headers: {
    "Content-Type": "application/x-www-form-urlencoded",
    language: "en",
  },
});

// Get JWT from localStorage
const getAccessToken = () => localStorage.getItem("loginjwt");
const getRefreshToken = () => localStorage.getItem("refreshjwt");

// Add JWT to headers
const addJwtToHeaders = () => {
  const token = getAccessToken();
  if (token) {
    return { Authorization: `Bearer ${token,
      refresh_token: refresh,
    }`,
      
     };
  }
  return {};
};

// Refresh token logic
const refreshToken = async () => {
  try {
    const refresh_token = getRefreshToken();
    if (!refresh_token) throw new Error("No refresh token found");

    const response = await axios.post(
      `${BASE_URL}/refresh_token`,
      {},
      {
        headers: {
          refresh_token: refresh_token,
        },
      }
    );

    const newAccessToken = response?.data?.data?.token;
    const newRefreshToken = response?.data?.data?.refresh_token;

    // Save new tokens
    localStorage.setItem("loginjwt", newAccessToken);
    localStorage.setItem("refreshjwt", newRefreshToken);

    return newAccessToken;
  } catch (error) {
    console.error("Token refresh failed:", error);
    localStorage.clear();
    window.location.href = "/login";
    return null;
  }
};

// Generic API call with refresh logic
const handleApiRequest = async (method:any, url:any, data = {}) => {
  try {
    const response = await apiClient({
      method,
      url,
      data,
      headers: addJwtToHeaders(),
    });
    return response;
  } catch (error) {
    if (error.response?.status === 401) {
      const newAccessToken = await refreshToken();
      if (newAccessToken) {
        return apiClient({
          method,
          url,
          data,
          headers: {
            ...addJwtToHeaders(),
            Authorization: `Bearer ${newAccessToken}`,
          },
        });
      }
    }
    throw error;
  }
};

// Exported functions
export const _post = (url:any, data = {}) => handleApiRequest("post", url, data);
export const getFunction = (url:any) => handleApiRequest("get", url);
export const _put = (url:any, data = {}) => handleApiRequest("put", url, data);
export const _delete = (url:any, data = {}) => handleApiRequest("delete", url, data);
export { refreshToken };
