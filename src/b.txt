import axios from "axios";

const BASE_URL =
  process.env.NEXT_PUBLIC_BASEAPI || "http://192.168.2.181:3000/admin";

// Create axios instance with base configuration
const apiClient = axios.create({
  baseURL: BASE_URL,
  headers: {
    language: "en",
    "Content-Type": "application/x-www-form-urlencoded",
  },
});

// Dynamic API call function
const apiCall = async ({
  method, // No default method
  endpoint,
  data = null,
  params = null,
  customHeaders = {},
}) => {
  // Check if method is provided
  if (!method) {
    throw new Error("HTTP method is required");
  }

  // Get JWT tokens from localStorage at the time of the request
  const jwt = localStorage.getItem("loginjwt");
  const refresh = localStorage.getItem("refreshjwt");

  try {
    // Configure the request
    const config = {
      method: method.toLowerCase(),
      url: endpoint,
      headers: {
        ...apiClient.defaults.headers,
        Authorizations: jwt ? `${jwt}` : undefined,
        refresh_token: refresh || undefined,
        ...customHeaders,
      },
      ...(data && { data }), // Add data if provided (for POST/PUT)
      ...(params && { params }), // Add query params if provided (for GET)
    };

    // Make the API call
    const response = await apiClient(config);

    // Return the response data
    return {
      success: true,
      data: response.data,
      status: response.status,
    };
  } catch (error) {
    // Handle errors
    console.error(`API ${method} request failed:`, error);

    // Return error details
    return {
      success: false,
      error: error.response?.data || error.message,
      status: error.response?.status || 500,
    };
  }
};

// Export the function
export default apiCall;

// Optional: Export specific method helpers
export const get = (endpoint, params, customHeaders) =>
  apiCall({ method: "get", endpoint, params, customHeaders });
export const post = (endpoint, data, customHeaders) =>
  apiCall({ method: "post", endpoint, data, customHeaders });
export const put = (endpoint, data, customHeaders) =>
  apiCall({ method: "put", endpoint, data, customHeaders });
export const del = (endpoint, customHeaders) =>
  apiCall({ method: "delete", endpoint, customHeaders });




// This will work
const fetchCategories = async () => {
  const response = await apiCall({
    method: "get", // Required
    endpoint: "/getcategories",
    params: { page: 1, limit: 10 },
  });
};

// This will throw an error
const invalidCall = async () => {
  const response = await apiCall({

  });

const fetchCategoriesShort = async () => {
